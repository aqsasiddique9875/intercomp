const backendOrigin = window.location.origin.includes('localhost') ? 'http://localhost:8080' : window.location.origin; let studentsData = []; let currentSkillFilter = 'All'; // Fetch Students from Backend async function loadStudents() { try { const res = await fetch(`${backendOrigin}/api/students`); if (res.ok) { const rawData = await res.json(); studentsData = rawData.map(u => ({ name: u.username || u.name || 'Anonymous Student', batch: u.graduationYear || '2026', skills: u.skills || [], score: u.trustScore || 100, avatar: u.avatar || `https://api.dicebear.com/7.x/adventurer/svg?seed=${encodeURIComponent(u.username || 'student')}` })); renderStudents(studentsData); } } catch (e) { console.error("Failed to load students:", e); } } // Render Student Cards function renderStudents(list) { const container = document.getElementById('studentList'); container.innerHTML = ''; if (list.length === 0) { container.innerHTML = `
No students found matching the criteria.
`; return; } list.forEach(student => { const skillsHtml = student.skills.map(s => ` ${s} `).join(''); const card = document.createElement('div'); card.className = "glass-card p-4 rounded-2xl flex items-start gap-3.5"; card.innerHTML = `

${student.name}

Batch ${student.batch} • Trust Score: ${student.score}%

Verified
${skillsHtml}
`; container.appendChild(card); }); } // Filter Logic function filterStudents() { const search = document.getElementById('searchInput').value.toLowerCase().trim(); const batch = document.getElementById('batchFilter').value; const filtered = studentsData.filter(student => { const matchesSearch = student.name.toLowerCase().includes(search) || student.skills.some(s => s.toLowerCase().includes(search)); const matchesBatch = (batch === 'All') || (student.batch === batch); const matchesSkill = (currentSkillFilter === 'All') || student.skills.includes(currentSkillFilter); return matchesSearch && matchesBatch && matchesSkill; }); renderStudents(filtered); } // Skill Filter Pill Click function selectSkillFilter(skill, btnElement) { currentSkillFilter = skill; // Update active style of pills document.querySelectorAll('.skill-pill-btn').forEach(btn => { btn.className = "skill-pill-btn shrink-0 px-3.5 py-1.5 rounded-full text-[10px] font-bold bg-white/40 text-outline border border-white/50 hover:bg-white/60 transition-all"; }); btnElement.className = "skill-pill-btn shrink-0 px-3.5 py-1.5 rounded-full text-[10px] font-bold bg-primary text-white shadow-sm transition-all"; filterStudents(); } // Initial Load loadStudents(); // Touch Feedback document.querySelectorAll('.glass-card, button').forEach(el => { el.addEventListener('touchstart', () => { el.style.transform = 'scale(0.97)'; el.style.transition = 'transform 0.15s ease-in-out'; }); el.addEventListener('touchend', () => { el.style.transform = 'scale(1)'; }); });
